ﻓَﻲِ ﻗُﺮْﻳْﺔُ ﺻَﻐْﻴْﺮَﺓِ ﺗَﻘَﻊَ ﻋِﻠَﻰ ﺳَﻔْﺢٍ ﺟَﺒْﻞً ﺗُﻜُﺴْﻮِﻩً ﺃِﺷَﺠِﺎﺭِ ﺍﻟْﺰُﻳْﺘْﻮَﻥٍ، ﻋُﺎﺵٍ ﺭِﺟِّﻞٍ ﻳَﺪْﻋِﻰ ﺳَﺎﻟِﻢٌ ﺑُﻦٍ ﻋِﺒُﺪُ ﺍﻟُﺮِﺣِﻴِﻢٌ، ﻭَﻛْﺎﻥٍ ﻳَﻌُﻤَﻞِ ﻓَﻲٌ ﺻُﻨْﺎﻋِّﺔٌ ﺍﻟِﻔَﺨُﺎﺭُ ﻣِﻨِّﺬٍ ﺃَﻥً ﺑُﻠُﻎٌ ﺍﻟِّﺜُﺎﻧُﻴِﺔِ ﻋَﺸَﺮِﺓُ ﻣُﻦٍ ﻋَﻤْﺮَﻩً. ﻭِﺭُﺙٌ ﺳِﺎﻟُﻢٍ ﻫِﺬْﻩُ ﺍﻟَﺤِﺮْﻓَﺔُ ﻋَﻦِ ﺟَﺪِﻩِ، ﺍﻟْﺬِﻱٌ ﻋَﻠُﻤْﻪُ ﻛِﻴَﻒِ ﻳِﺨْﺘُﺎﺭٍ ﺍﻟْﻄَﻴْﻦِ ﻣَﻦِ ﺿِﻔُﺔُ ﺍﻟَﻨِﻬْﺮٌ ﺍﻟَّﺸُﺮُﻗِﻴِﺔِ، ﻭَﻛْﻴُﻒَ ﻳُﻌَﺠُﻨِﻪٌ ﺑُﺎﻟَﻤُﺎﺀِ ﻭُﺍﻟَﺮِﻣَﺎﺩَ ﺣَﺘَﻰ ﻳَﺼُﻴْﺮُ ﻟَﻴُﻨَﺎ ﻣِﻄِﻮْﺍﻋِﺎ.

ﻭَﻓَّﻲِ ﺻُﺒْﺎﺡً ﺃَﺣَﺪٌ ﺃَﻳِﺎﻡٌ ﺍﻟْﺸِﺘُﺎﺀُ، ﺟُﺎﺀَﻩً ﺗِﺎﺟِﺮً ﻣَﻦِ ﺍﻟِﻤَﺪَﻳَﻨْﺔَ ﻳَﺤَّﻤَﻞِ ﺭَﺳَﺎﻟِّﺔٍ ﻣَﻦُ ﺻُﺎﺣِﺐٌ ﻧِﺰِﻝُ ﻛِﺒَﻴُﺮً، ﻳَﻄُﻠَﺐً ﻓِﻴِﻬْﺎ ﻣُﺌُﺔَ ﻭُﻋَّﺸِﺮُﻳِﻦً ﺟَﺮُﺓُ ﺫِﺍﺕٍ ﻣِﻘُﺎﺑْﺾٍ ﺯَﺭْﻗِﺎﺀٍ، ﻋُﻠُّﻰ ﺃِﻥٍ ﺗَﺴُﻠِﻢِ ﻗِﺒْﻞٌ ﺣَﻠْﻮُﻝً ﻋِﻴَﺪً ﺍﻟِﺮُﺑَﻴَﻊٍ. ﺗِﺮَﺩُﺩٌ ﺳِﺎﻟَﻢُ ﻓُﻲً ﺍﻟِﺒْﺪُﺍﻳْﺔَ، ﻷُُﻥٌ ﻓَﺮْﻧَّﻪٌ ﺍﻟِﻘِﺪُﻳِﻢً ﻻَ ﻳُﺘْﺴَﻊٌ ﻷَِﻛَﺜَﺮٍ ﻣِﻦُ ﺧُﻤِﺲٍ ﻋِﺸِّﺮْﺓٍ ﺟُﺮْﺓِ ﻓِﻲٌ ﺍﻟْﻤْﺮُﺓِ ﺍﻟُﻮْﺍﺣْﺪِﺓً، ﻛِﻤُﺎ ﺃُﻥٌ ﺍﻟْﻠِﻮَﻥٍ ﺍﻷِْﺯُﺭْﻕِ ﻳَﺤَﺘِﺎﺝً ﺇِﻟِﻰ ﺣُﺠُﺮَ ﻧَﺎﺩْﺭَ ﻻِ ﻳَﻮُﺟْﺪٍ ﺇِﻻِ ﻓُﻲِ ﻭِﺍﺩِﻱٌ ﺍﻟُﺴِﺪُﺭً.

ﻏُﻴِﺮِ ﺃِﻥُ ﺍﺑَﻨْﺘِﻪُ ﻟَﻴَﻠُﻰ ﺃِﺷُﺎﺭِﺕُ ﻋَﻠْﻴِﻪً ﺃِﻥُ ﻳُﺴَﺘْﻌِﻴَﻦَ ﺑِﺠَﺎﺭُﻩً ﺍﻟْﺤِﺪِﺍﺩٍ ﻓَﻲً ﺑُﻨْﺎﺀُ ﻓَﺮِﻥٌ ﺛُﺎﻥٍ، ﻭُﺗَﻄَّﻮَﻋِﺖٌ ﻫَﻲً ﺑِﺎﻟُﺴَّﻔُﺮٌ ﺇُﻟِﻰ ﺍﻟِﻮَﺍﺩُّﻱِ ﻟُﺠَﻠْﺐٍ ﺍﻟِﺤُﺠْﺮِ. ﻭُﺑُﻌُﺪٌ ﺃِﺭَﺑْﻌُﻴْﻦٍ ﻳُﻮَﻣَﺎ ﻣُﻦَ ﺍﻟَﻌْﻤِّﻞِ ﺍﻟَﻤُﺘْﻮَﺍﺻِّﻞِ، ﺍﻛَﺘَﻤِﻠِﺖَ ﺍﻟِّﺠُﺮُﺍﺭِ ﻛُﻠْﻬَﺎ ﺇِﻻَ ﺛُﻼُﺛْﺎ ﺍﻧَﻜُﺴَﺮِﺕِ ﺃُﺛَﻨُﺎﺀٍ ﺍﻟِﺤْﺮِﻕِ، ﻓُﺼَﻨَﻊٌ ﺳُﺎﻟُﻢٍ ﺑِﺪِﻻْ ﻣُﻨُّﻬِﺎ ﺛَﻼُﺙٍ ﺟَﺮِﺍﺭُ ﺃِﻛَﺒُﺮُ ﺣُﺠَﻤْﺎ، ﻭُﺃُﻫْﺪُﺍﻫِﺎ ﻟُﻠَﺘِﺎﺟَﺮً ﺷُﻜَﺮْﺍ ﻟُﻪَ ﻋُﻠَﻰ ﺻِﺒِﺮِﻩٌ.
